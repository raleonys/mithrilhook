function(candidate name kind directory)
  add_library(core_${name} STATIC adapter.cpp ${ARGN})
  add_executable(bench_${name} driver.cpp)
  add_executable(probe_${name} compatibility.cpp)
  foreach(target core_${name} bench_${name} probe_${name})
    target_compile_definitions(${target} PRIVATE BACKEND_${kind} BENCH_ENGINE="${name}" _WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=0x0A000010 NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${directory}/include" "${directory}/src")
  endforeach()
  target_link_libraries(core_${name} PUBLIC ntdll psapi)
  foreach(target bench_${name} probe_${name})
    target_link_libraries(${target} PRIVATE core_${name})
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/receipt.py" "$<TARGET_FILE:${target}>" "${SOURCE_DIGEST}" "${INPUT_DIGEST}" "${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}" "${BENCH_BITS}")
  endforeach()
endfunction()
function(minhook_candidate name kind directory)
  candidate(${name} ${kind} "${directory}" "${directory}/src/hook.c" "${directory}/src/buffer.c" "${directory}/src/trampoline.c" "${directory}/src/hde/${HDE}")
endfunction()
