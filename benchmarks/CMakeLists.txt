cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(MithrilHookBenchmarks LANGUAGES C CXX)
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Benchmark configuration" FORCE)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Performance evidence requires native Windows MSVC")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(SOURCES "${REPO_ROOT}/build/bench-sources")
set(PREPARED "${REPO_ROOT}/build/bench-prepared")
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_C_FLAGS_RELEASE "/O2 /Ob2 /DNDEBUG")
set(CMAKE_CXX_FLAGS_RELEASE "/O2 /Ob2 /DNDEBUG")
add_compile_options(/W3 /Gy /Oi "$<$<COMPILE_LANGUAGE:C>:/std:clatest>")
file(SHA256 "${REPO_ROOT}/metadata/SOURCE_MANIFEST.json" SOURCE_DIGEST)
file(SHA256 "${PREPARED}/inputs.json" INPUT_DIGEST)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${REPO_ROOT}/metadata/SOURCE_MANIFEST.json" "${PREPARED}/inputs.json")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/identity.h"
  "#define BENCH_SOURCE_DIGEST \"${SOURCE_DIGEST}\"\n#define BENCH_INPUT_DIGEST \"${INPUT_DIGEST}\"\n#define BENCH_COMPILER \"${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}\"\n")
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(HDE hde64.c)
else()
  set(HDE hde32.c)
endif()
function(candidate name kind directory)
  add_executable(bench_${name} driver.cpp adapter.cpp ${ARGN})
  target_compile_definitions(bench_${name} PRIVATE BACKEND_${kind} BENCH_ENGINE="${name}" _WIN32_WINNT=0x0601 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(bench_${name} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${directory}/include" "${directory}/src")
  target_link_libraries(bench_${name} PRIVATE ntdll psapi)
  add_custom_command(TARGET bench_${name} POST_BUILD
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/receipt.py" "$<TARGET_FILE:bench_${name}>" "${SOURCE_DIGEST}" "${INPUT_DIGEST}" "${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}")
endfunction()
function(minhook_candidate name kind directory)
  candidate(${name} ${kind} "${directory}" "${directory}/src/hook.c" "${directory}/src/buffer.c" "${directory}/src/trampoline.c" "${directory}/src/hde/${HDE}")
endfunction()
minhook_candidate(mithrilhook MTH "${REPO_ROOT}")
minhook_candidate(official MH "${SOURCES}/official")
minhook_candidate(patched MH "${PREPARED}/patched")
target_compile_definitions(bench_patched PRIVATE MH_SAFE_FREEZE=1)
minhook_candidate(m417z MH "${SOURCES}/m417z")
set(DETOUR_CPP)
foreach(file detours modules disasm image creatwth disolx86 disolx64 disolia64 disolarm disolarm64)
  list(APPEND DETOUR_CPP "${SOURCES}/detours/src/${file}.cpp")
endforeach()
candidate(detours DETOURS "${SOURCES}/detours" ${DETOUR_CPP})
set(SLIM_C)
set(MHD_C)
foreach(file Disassembler InlineHook Instruction Memory Thread Trampoline Transaction)
  list(APPEND SLIM_C "${SOURCES}/slim/Source/KNSoft.SlimDetours/${file}.c")
  list(APPEND MHD_C "${SOURCES}/mhd/src/SlimDetours/${file}.c")
endforeach()
candidate(slim SLIM "${SOURCES}/slim/Source/KNSoft.SlimDetours" ${SLIM_C})
target_include_directories(bench_slim PRIVATE "${SOURCES}/slim/Source/KNSoft.SlimDetours" "${SOURCES}/mhd/src/phnt")
target_compile_options(bench_slim PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/nt_headers.h")
candidate(mhd MHD "${SOURCES}/mhd" "${SOURCES}/mhd/src/MinHook.c" ${MHD_C})
target_include_directories(bench_mhd PRIVATE "${SOURCES}/mhd/src/phnt")
