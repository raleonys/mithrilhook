cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(MithrilHookInstalledConsumer LANGUAGES CXX)
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(MithrilHook_ALLOW_PRERELEASE OFF)
find_package(MithrilHook CONFIG REQUIRED)
if(MithrilHook_PRERELEASE)
  set(MithrilHook_ALLOW_PRERELEASE OFF)
  find_package(MithrilHook 1.0 CONFIG QUIET)
  if(MithrilHook_FOUND)
    message(FATAL_ERROR "A numeric stable request silently accepted a prerelease package")
  endif()
  set(MithrilHook_ALLOW_PRERELEASE ON)
endif()
find_package(MithrilHook 1.0.5 EXACT CONFIG REQUIRED)
find_package(MithrilHook 1.0 CONFIG REQUIRED)
find_package(MithrilHook 2.0 CONFIG QUIET)
if(MithrilHook_FOUND)
  message(FATAL_ERROR "An incompatible major request accepted this package")
endif()
find_package(MithrilHook 1.0.5 EXACT CONFIG REQUIRED)
set(expected_version "${MithrilHook_VERSION_NUMERIC}")
if(MithrilHook_PRERELEASE)
  string(APPEND expected_version "-${MithrilHook_PRERELEASE}")
endif()
if(NOT MithrilHook_VERSION STREQUAL expected_version)
  message(FATAL_ERROR "Installed package numeric/full version metadata differs")
endif()
add_executable(installed_consumer consumer.cpp)
target_compile_features(installed_consumer PRIVATE cxx_std_17)
target_link_libraries(installed_consumer PRIVATE MithrilHook::mithrilhook)
get_target_property(library_type MithrilHook::mithrilhook TYPE)
if(library_type STREQUAL "SHARED_LIBRARY")
  add_custom_command(TARGET installed_consumer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:MithrilHook::mithrilhook> $<TARGET_FILE_DIR:installed_consumer>)
endif()
enable_testing()
add_test(NAME installed_package_consumer COMMAND installed_consumer)
set_tests_properties(installed_package_consumer PROPERTIES TIMEOUT 45)
