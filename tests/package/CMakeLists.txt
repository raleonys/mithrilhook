cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(MithrilHookInstalledConsumer LANGUAGES CXX)
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(MithrilHook 0.1 CONFIG REQUIRED)
add_executable(installed_consumer consumer.cpp)
target_compile_features(installed_consumer PRIVATE cxx_std_17)
target_link_libraries(installed_consumer PRIVATE MithrilHook::mithrilhook)
get_target_property(library_type MithrilHook::mithrilhook TYPE)
if(library_type STREQUAL "SHARED_LIBRARY")
  add_custom_command(TARGET installed_consumer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:MithrilHook::mithrilhook> $<TARGET_FILE_DIR:installed_consumer>)
endif()
enable_testing()
add_test(NAME installed_package_consumer COMMAND installed_consumer)
set_tests_properties(installed_package_consumer PROPERTIES TIMEOUT 45)
