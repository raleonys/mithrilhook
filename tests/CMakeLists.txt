function(mth_fixture name source)
  add_executable(${name} ${source} ${MTH_PRIVATE})
  target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/src ${PROJECT_SOURCE_DIR}/include)
  target_compile_definitions(${name} PRIVATE _WIN32_WINNT=0x0601)
  target_compile_features(${name} PRIVATE c_std_17)
  add_test(NAME ${name} COMMAND ${name} ${ARGN})
  set_tests_properties(${name} PROPERTIES TIMEOUT 45)
endfunction()
function(mth_linked name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE mithrilhook)
  target_compile_features(${name} PRIVATE c_std_17)
  set_target_properties(${name} PROPERTIES CXX_STANDARD 17)
  target_compile_definitions(${name} PRIVATE _WIN32_WINNT=0x0601 NOMINMAX)
  if(BUILD_SHARED_LIBS)
    add_custom_command(TARGET ${name} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:mithrilhook> $<TARGET_FILE_DIR:${name}>)
  endif()
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 45)
endfunction()
